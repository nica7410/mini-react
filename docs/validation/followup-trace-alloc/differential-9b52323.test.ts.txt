function rng(seed: number) { let s = seed >>> 0 || 1; return () => { s ^= s << 13; s ^= s >>> 17; s ^= s << 5; return (s >>> 0) / 4294967296; }; }
const strip = (v: unknown): unknown => JSON.parse(JSON.stringify(v, (k, x) => typeof x === 'function' ? `fn:${x.name}` : x));
function Row(p: { label: string; n: number }) { return h('li', { attrs: { title: p.label } }, p.label, p.n % 2 ? h('b', null, 'x') : null); }
function tree(r: () => number, depth: number): any {
  const n = Math.floor(r() * 5);
  const keyedAll = r() < .5;
  const kids = Array.from({ length: n }, (_, i) => {
    const t = r();
    const key = keyedAll ? `k${Math.floor(r() * 6)}` : r() < .3 ? `k${i}` : undefined;
    if (t < .2) return `t${i}`;
    if (t < .3) return r() < .5 ? null : [`a${i}`, [h('i', { key }, 'n')]];
    if (t < .45) return h(Row, { key, label: `L${i}`, n: i });
    const tags = ['div', 'span', 'p', 'input', 'select', 'option', 'form', 'textarea', 'bogus', 'img'];
    const tag = tags[Math.floor(r() * tags.length)];
    const attrs: any = r() < .5 ? { title: 'x', ...(r() < .1 ? { onclick: 'y' } : {}), ...(tag === 'input' && r() < .5 ? { type: r() < .5 ? 'radio' : 'checkbox', name: 'g' } : {}) } : undefined;
    const form = ['input', 'select', 'textarea'].includes(tag) && r() < .7 ? (r() < .5 ? { mode: 'controlled', value: 'v', checked: r() < .5, ...(tag === 'select' ? { selectedIndices: [0] } : {}) } : { mode: 'uncontrolled', defaultValue: 'd' }) : undefined;
    return h(tag, { key, attrs, form, style: r() < .2 ? { color: r() < .8 ? 'red' : 'expression(1)' } : undefined, events: r() < .3 ? { click: () => undefined } : undefined },
      ...(depth > 0 && !['input', 'img'].includes(tag) ? [tree(r, depth - 1)] : []));
  });
  return h('div', null, ...kids);
}
